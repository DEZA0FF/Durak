    <script>
        const tg = window.Telegram.WebApp;
        tg.expand();

        const suits = ['♥', '♦', '♣', '♠'];
        const values = ['6', '7', '8', '9', '10', 'J', 'Q', 'K', 'A'];
        const rankValue = {'6':6, '7':7, '8':8, '9':9, '10':10, 'J':11, 'Q':12, 'K':13, 'A':14};

        let deck = [];
        let trumpSuit = '';
        let players = [
            { id: 0, name: 'Вы', hand: [], isBot: false },
            { id: 1, name: 'Бот 1', hand: [], isBot: true },
            { id: 2, name: 'Бот 2', hand: [], isBot: true }
        ];
        
        let table = []; // Массив пар [{attack: card, defense: card/null}]
        let attackerIdx = 0; // Кто ходит
        let defenderIdx = 1; // Кто отбивается
        let gamePhase = 'init'; // 'attack', 'defense', 'transfer'
        let selectedCards = [];

        function initGame() {
            deck = [];
            suits.forEach(s => {
                values.forEach(v => {
                    deck.push({ suit: s, value: v, rank: rankValue[v] });
                });
            });
            // Перемешивание
            deck.sort(() => Math.random() - 0.5);

            // Раздача по 6 карт
            players.forEach(p => {
                p.hand = deck.splice(0, 6);
                sortHand(p.hand);
            });

            // Козырь
            const trumpCard = deck[deck.length - 1];
            trumpSuit = trumpCard.suit;

            // Определение первого ходящего (у кого младший козырь)
            let minTrump = 15;
            let firstPlayer = 0;
            players.forEach((p, pIdx) => {
                p.hand.forEach(c => {
                    if (c.suit === trumpSuit && c.rank < minTrump) {
                        minTrump = c.rank;
                        firstPlayer = pIdx;
                    }
                });
            });

            attackerIdx = firstPlayer;
            defenderIdx = (attackerIdx + 1) % 3;

            renderTrump(trumpCard);
            startTurn();
        }

        function sortHand(hand) {
            hand.sort((a, b) => {
                if (a.suit === trumpSuit && b.suit !== trumpSuit) return 1;
                if (b.suit === trumpSuit && a.suit !== trumpSuit) return -1;
                return a.rank - b.rank;
            });
        }

        function renderTrump(card) {
            const container = document.getElementById('trump-container');
            container.innerHTML = createCardHTML(card);
        }

        function createCardHTML(card, isHidden = false) {
            if (isHidden) return `<div class="card back"></div>`;
            const isRed = card.suit === '♥' || card.suit === '♦';
            return `<div class="card ${isRed ? 'red' : ''}" onclick="selectCard('${card.value}', '${card.suit}')">
                <div>${card.value}</div>
                <div style="align-self: center; font-size: 18px;">${card.suit}</div>
                <div style="align-self: flex-end;">${card.value}</div>
            </div>`;
        }

        function selectCard(val, suit) {
            if (players[0].isBot || gamePhase === 'defense' && defenderIdx !== 0) {
                // Если ход игрока-защитника, кликом выбираем одну карту для ответа
                if (defenderIdx === 0 && table.length > 0) {
                    const unbeatPair = table.find(p => !p.defense);
                    if (unbeatPair) {
                        const cardIdx = players[0].hand.findIndex(c => c.value === val && c.suit === suit);
                        if (cardIdx !== -1) {
                            const card = players[0].hand[cardIdx];
                            if (canBeat(unbeatPair.attack, card, trumpSuit)) {
                                players[0].hand.splice(cardIdx, 1);
                                unbeatPair.defense = card;
                                selectedCards = [];
                                updateUI();
                                checkAttackContinuation();
                            } else {
                                alert("Этой картой нельзя побить!");
                            }
                        }
                    }
                }
                return;
            }

            // Логика выбора карт для атаки или подкидывания
            const idx = selectedCards.findIndex(c => c.value === val && c.suit === suit);
            if (idx > -1) {
                selectedCards.splice(idx, 1);
            } else {
                // Проверка на соответствие номиналам на столе (если стол не пустой)
                if (table.length > 0) {
                    const tableValues = [];
                    table.forEach(p => {
                        tableValues.push(p.attack.value);
                        if (p.defense) tableValues.push(p.defense.value);
                    });
                    if (!tableValues.includes(val)) {
                        alert("Такую карту нельзя подкидывать!");
                        return;
                    }
                }
                // Проверка на перевод (если фаза перевода доступна)
                if (gamePhase === 'transfer' && table.length === 1 && table[0].attack.value === val) {
                    // Разрешаем перевод
                }
                selectedCards.push({ value: val, suit: suit });
            }
            updateUI();
        }

        function canBeat(attCard, defCard, trump) {
            if (defCard.suit === trump && attCard.suit !== trump) return true;
            if (defCard.suit === attCard.suit && defCard.rank > attCard.rank) return true;
            return false;
        }

        function startTurn() {
            table = [];
            selectedCards = [];
            gamePhase = 'attack';
            updateUI();

            if (players[attackerIdx].isBot) {
                setTimeout(botAttack, 1000);
            } else {
                setStatus(`Ваш ход (атака на ${players[defenderIdx].name})`);
            }
        }

        function botAttack() {
            const bot = players[attackerIdx];
            if (bot.hand.length === 0) { endTurnClean(); return; }
            
            // Бот ходит самой младшей не козырной картой
            let cardToAttack = bot.hand.find(c => c.suit !== trumpSuit) || bot.hand[0];
            bot.hand = bot.hand.filter(c => !(c.value === cardToAttack.value && c.suit === cardToAttack.suit));
            
            table.push({ attack: cardToAttack, defense: null });
            gamePhase = 'defense';
            updateUI();

            setTimeout(botDefendOrTransfer, 1000);
        }

        function botDefendOrTransfer() {
            const defender = players[defenderIdx];
            const unbeatPair = table[table.length - 1];

            // Проверка на перевод (если бот-защитник может и хочет перевести)
            let canTransfer = false;
            if (table.length === 1 && gamePhase === 'transfer') {
                const transferCard = defender.hand.find(c => c.value === unbeatPair.attack.value);
                if (transferCard && defender.hand.length > 2) {
                    // Переводим
                    defender.hand = defender.hand.filter(c => !(c.value === transferCard.value && c.suit === transferCard.suit));
                    table.push({ attack: transferCard, defense: null });
                    attackerIdx = defenderIdx;
                    defenderIdx = (defenderIdx + 1) % 3;
                    setStatus(`${defender.name} переводит карты на ${players[defenderIdx].name}!`);
                    updateUI();
                    setTimeout(botDefendOrTransfer, 1500);
                    return;
                }
            }

            // Обычная защита бота
            const defCard = defender.hand.find(c => canBeat(unbeatPair.attack, c, trumpSuit));
            if (defCard) {
                defender.hand = defender.hand.filter(c => !(c.value === defCard.value && c.suit === defCard.suit));
                unbeatPair.defense = defCard;
                setStatus(`${defender.name} отбился.`);
                updateUI();
                setTimeout(botSubAttackOrEnd, 1000);
            } else {
                // Бот берет карты
                setStatus(`${defender.name} берет карты!`);
                setTimeout(() => {
                    table.forEach(p => {
                        defender.hand.push(p.attack);
                        if (p.defense) defender.hand.push(p.defense);
                    });
                    sortHand(defender.hand);
                    table = [];
                    // Пропуск хода отбивающегося, ход переходит к следующему за ним
                    attackerIdx = (defenderIdx + 1) % 3;
                    defenderIdx = (attackerIdx + 1) % 3;
                    refillHands();
                    startTurn();
                }, 1500);
            }
        }

        function botSubAttackOrEnd() {
            // Боты могут подкидывать
            const attacker = players[attackerIdx];
            const allValues = [];
            table.forEach(p => { allValues.push(p.attack.value); if(p.defense) allValues.push(p.defense.value); });

            const subCard = attacker.hand.find(c => allValues.includes(c.value));
            if (subCard && table.length < 6) {
                attacker.hand = attacker.hand.filter(c => !(c.value === subCard.value && c.suit === subCard.suit));
                table.push({ attack: subCard, defense: null });
                gamePhase = 'defense';
                setStatus(`${attacker.name} подкидывает карту.`);
                updateUI();
                setTimeout(botDefendOrTransfer, 1000);
            } else {
                // Отбой
                setStatus(`Карты побиты! Бито.`);
                setTimeout(endTurnClean, 1500);
            }
        }

        function endTurnClean() {
            table = [];
            attackerIdx = (attackerIdx + 1) % 3;
            defenderIdx = (attackerIdx + 1) % 3;
            refillHands();
            startTurn();
        }

        function refillHands() {
            players.forEach(p => {
                while (p.hand.length < 6 && deck.length > 0) {
                    p.hand.push(deck.pop());
                }
                sortHand(p.hand);
            });
        }

        function handleAction() {
            // Кнопка действий игрока (Подкинуть, Бито или Взять)
            if (attackerIdx === 0 && gamePhase === 'attack') {
                if (selectedCards.length === 0) {
                    // Завершение атаки / Бито
                    setStatus("Вы записали «Бито».");
                    setTimeout(endTurnClean, 1000);
                } else {
                    // Подкидывание карт игроком
                    selectedCards.forEach(sc => {
                        const idx = players[0].hand.findIndex(c => c.value === sc.value && c.suit === sc.suit);
                        if (idx !== -1) {
                            players[0].hand.splice(idx, 1);
                            table.push({ attack: sc, defense: null });
                        }
                    });
                    selectedCards = [];
                    gamePhase = 'defense';
                    updateUI();
                    // Ход переходит к боту-защитнику
                    setTimeout(botDefendOrTransfer, 1000);
                }
            } else if (defenderIdx === 0) {
                // Игрок берет карты
                table.forEach(p => {
                    players[0].hand.push(p.attack);
                    if (p.defense) players[0].hand.push(p.defense);
                });
                sortHand(players[0].hand);
                table = [];
                attackerIdx = (defenderIdx + 1) % 3;
                defenderIdx = (attackerIdx + 1) % 3;
                refillHands();
                startTurn();
            }
        }

        function updateUI() {
            document.getElementById('deck-count').innerText = `Колода: ${deck.length}`;
            document.getElementById('bot1-info').innerText = `${players[1].name} (Карт: ${players[1].hand.length})`;
            document.getElementById('bot2-info').innerText = `${players[2].name} (Карт: ${players[2].hand.length})`;

            // Отрисовка поля боя
            const field = document.getElementById('battle-field');
            field.innerHTML = '';
            table.forEach(pair => {
                const pairDiv = document.createElement('div');
                pairDiv.className = 'pair';
                pairDiv.innerHTML = createCardHTML(pair.attack) + (pair.defense ? createCardHTML(pair.defense) : '<div class="card back" style="opacity:0.5"></div>');
                field.appendChild(pairDiv);
            });

            // Отрисовка руки игрока
            const handDiv = document.getElementById('player-hand');
            handDiv.innerHTML = '';
            players[0].hand.forEach(c => {
                const isSelected = selectedCards.some(sc => sc.value === c.value && sc.suit === c.suit);
                const isRed = c.suit === '♥' || c.suit === '♦';
                const cardEl = document.createElement('div');
                cardEl.className = `card ${isRed ? 'red' : ''} ${isSelected ? 'selected' : ''}`;
                cardEl.innerHTML = `<div>${c.value}</div><div style="align-self: center; font-size: 18px;">${c.suit}</div><div style="align-self: flex-end;">${c.value}</div>`;
                cardEl.onclick = () => selectCard(c.value, c.suit);
                handDiv.appendChild(cardEl);
            });

            // Настройка кнопки действий
            const btn = document.getElementById('action-btn');
            btn.disabled = false;
            if (attackerIdx === 0 && table.length === 0) {
                btn.innerText = "Завершить ход (Бито)";
            } else if (attackerIdx === 0 && table.length > 0) {
                btn.innerText = selectedCards.length > 0 ? "Подкинуть" : "Бито / Хватит";
            } else if (defenderIdx === 0) {
                btn.innerText = "Беру";
            } else {
                btn.disabled = true;
                btn.innerText = "Ход противника...";
            }
        }
